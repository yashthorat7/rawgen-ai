import { notFound } from "next/navigation";
import { Header } from "@/components/layout/Header";
import { AiSuggestor } from "@/components/intro/AiIntro";
import { PlatformProvider } from "@/lib/PlatformContext";
import { ProfileGrid } from "@/components/profile/ProfileGrid";
import { Intro } from "@/components/intro/Intro";
import { TagStrip } from "@/components/intro/TagStrip";
import { SkillGrid } from "@/components/skill/SkillGrid";
import { ProjectGrid } from "@/components/project/ProjectGrid";
import { CertificateGrid } from "@/components/certificate/CertificateGrid";
import { Footer } from "@/components/layout/Footer";
import { api } from "@/convex/_generated/api";
import { ConvexHttpClient } from "convex/browser";

interface PublicProfilePageProps {
  params: Promise<{
    username: string;
  }>;
}

const convex = new ConvexHttpClient(process.env.NEXT_PUBLIC_CONVEX_URL!);

export default async function PublicProfilePage({ params }: PublicProfilePageProps) {
  const { username } = await params;

  // Username validation and user existence check
  if (!username) {
    notFound();
  }

  // Check if user exists
  const userData = await convex.query(api.users.getUser, { username });
  if (!userData) {
    notFound();
  }

  return (
    <PlatformProvider isPublicView username={username}>
      <div className="min-h-screen flex flex-col relative">
        {/* Fixed Gradient Background */}
        <div className="fixed inset-0 bg-gradient-to-b from-transparent to-primary-100/50 pointer-events-none" />
        
        {/* Content */}
        <Header />
        <main className="flex-1 relative dotted-background">
          <div className="p-4 mt-4 mx-auto w-full lg:w-[90%]">
            <Intro />
            <AiSuggestor />
            <TagStrip />
            <ProfileGrid isPublicView />
            <SkillGrid />
            <ProjectGrid />
            <CertificateGrid />
          </div>
        </main>
        <Footer />
      </div>
    </PlatformProvider>
  );
}
